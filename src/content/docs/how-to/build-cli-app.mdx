```boba
import std:env
import std:os

fn main() {
    let args = env.args()
    
    if args.len() < 2 {
        print_usage()
        os.exit(1)
    }
    
    let command = args[1]
    match command {
        "help" => print_usage(),
        "process" => handle_process_command(args),
        _ => {
            print(f"Unknown command: {command}")
            print_usage()
            os.exit(1)
        }
    }
}

fn print_usage() {
    print("My CLI Tool v1.0")
    print("")
    print("Usage:")
    print("  my-cli-tool help                    Show this help")
    print("  my-cli-tool process <file>          Process a file")
    print("  my-cli-tool process <file> --output <out>  Process with custom output")
    print("")
}
```

## Step 3: Parse Command Arguments

```boba
struct ProcessOptions {
    input_file: string,
    output_file: Option<string>,
    verbose: boolean
}

fn handle_process_command(args: string[]) {
    let options = parse_process_args(args)
    match options {
        Ok(opts) => run_process(opts),
        Err(error) => {
            print(f"Error: {error}")
            os.exit(1)
        }
    }
}

fn parse_process_args(args: string[]): Result<ProcessOptions, string> {
    if args.len() < 3 {
        return Err("Missing input file. Usage: process <file>")
    }
    
    let input_file = args[2]
    var output_file: Option<string> = None
    var verbose = false
    
    // Parse optional flags
    var i = 3
    while i < args.len() {
        match args[i] {
            "--output" => {
                if i + 1 >= args.len() {
                    return Err("--output requires a filename")
                }
                output_file = Some(args[i + 1])
                i += 2
            },
            "--verbose" => {
                verbose = true
                i += 1
            },
            _ => {
                return Err(f"Unknown option: {args[i]}")
            }
        }
    }
    
    return Ok(ProcessOptions {
        input_file: input_file,
        output_file: output_file,
        verbose: verbose
    })
}
```

## Step 4: Implement File Processing

```boba
import std:fs
import std:string

fn run_process(options: ProcessOptions) {
    if options.verbose {
        print(f"Processing file: {options.input_file}")
    }
    
    // Read input file
    let content = fs.read(options.input_file)
    match content {
        Ok(text) => {
            let processed = process_content(text)
            save_output(processed, options)
        },
        Err(error) => {
            print(f"Failed to read {options.input_file}: {error}")
            os.exit(1)
        }
    }
}

fn process_content(content: string): string {
    // Example processing: count lines and words
    let lines = content.split("\n")
    let word_count = content
        .split_whitespace()
        .len()
    
    let result = string.join([
        f"File Analysis Report",
        f"=====================",
        f"Lines: {lines.len()}",
        f"Words: {word_count}",
        f"Characters: {content.len()}",
        f"",
        f"Content Preview:",
        f"---------------"
    ], "\n")
    
    // Include first few lines as preview
    let preview_lines = lines.take(5)
    let preview = string.join(preview_lines, "\n")
    
    return result + "\n" + preview
}

fn save_output(processed: string, options: ProcessOptions) {
    let output_path = options.output_file.unwrap_or("output.txt")
    
    let write_result = fs.write(path: output_path, content: processed)
    match write_result {
        Ok(_) => {
            if options.verbose {
                print(f"Results saved to: {output_path}")
            } else {
                print(f"Processed! Output saved to {output_path}")
            }
        },
        Err(error) => {
            print(f"Failed to write output: {error}")
            os.exit(1)
        }
    }
}
```

```boba
fn validate_input_file(path: string): Result<(), string> {
    // Check if file exists
    if !fs.exists(path) {
        return Err(f"File not found: {path}")
    }
    
    // Check if it's actually a file (not a directory)
    let metadata = fs.metadata(path)?
    if metadata.is_directory {
        return Err(f"Path is a directory, not a file: {path}")
    }
    
    // Check if file is readable
    if !metadata.readable {
        return Err(f"File is not readable: {path}")
    }
    
    return Ok(())
}
```

```boba
fn process_large_file(content: string, verbose: boolean) {
    let lines = content.split("\n")
    let total_lines = lines.len()
    
    if verbose {
        print(f"Processing {total_lines} lines...")
    }
    
    var processed_lines: string[] = []
    for (index, line) in lines.enumerate() {
        // Simulate processing work
        let processed_line = line.trim().to_upper()
        processed_lines.push(processed_line)
        
        // Show progress every 1000 lines
        if verbose && (index + 1) % 1000 == 0 {
            let percent = ((index + 1) * 100) / total_lines
            print(f"Progress: {percent}% ({index + 1}/{total_lines})")
        }
    }
    
    return string.join(processed_lines, "\n")
```

### Configuration Files

```boba
import std:json

struct Config {
    max_file_size: int,
    output_format: string,
    default_output_dir: string
}

fn load_config(): Config {
    let config_path = "config.json"
    let default_config = Config {
        max_file_size: 10_000_000,  // 10MB
        output_format: "txt",
        default_output_dir: "output"
    }
    
    if !fs.exists(config_path) {
        return default_config
    }
    
    let content = fs.read(config_path)
    match content {
        Ok(json_text) => {
            let parsed = json.parse<Config>(json_text)
            match parsed {
                Ok(config) => config,
                Err(_) => {
                    print("Warning: Invalid config file, using defaults")
                    default_config
                }
            }
        },
        Err(_) => default_config
    }
}
```
