## Step 9: Error Handling Best Practices

```boba
enum JsonError {
    ParseError(string),
    ValidationError(string),
    MissingField(string),
    InvalidFormat(string)
}

fn robust_json_parsing(json_string: string): Result<User[], JsonError> {
    // Try to parse as array first
    let parse_result = json.parse<User[]>(json_string)
    match parse_result {
        Ok(users) => {
            // Validate each user and collect errors
            let validation_result = validate_all_users(users)
            match validation_result {
                Ok(valid_users) => Ok(valid_users),
                Err(error) => Err(JsonError.ValidationError(error))
            }
        },
        Err(error) => {
            // Try parsing as single user
            let single_parse = json.parse<User>(json_string)
            match single_parse {
                Ok(user) => Ok([user]),
                Err(_) => Err(JsonError.ParseError(error))
            }
        }
    }
}

fn validate_all_users(users: User[]): Result<User[], string> {
    var errors: string[] = []
    var valid_users: User[] = []
    
    for user in users {
        match validate_user(user) {
            Ok(valid_user) => valid_users.push(valid_user),
            Err(error) => errors.push(f"User {user.id}: {error}")
        }
    }
    
    if !errors.is_empty() {
        return Err(string.join(errors, "; "))
    }
    
    return Ok(valid_users)
}
```
