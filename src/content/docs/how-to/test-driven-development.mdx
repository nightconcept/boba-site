## Goal

Learn to use Test-Driven Development (TDD) effectively in Boba, writing tests before implementation to drive better design and catch bugs early.

## Prerequisites

## TDD Cycle: Red-Green-Refactor

1. **Red**: Write a failing test
2. **Green**: Write minimal code to make it pass
3. **Refactor**: Improve the code while keeping tests passing

## Step 6: Test-Drive a More Complex Feature

Let's create a calculator that maintains state:

```boba
struct Calculator {
    current_value: float,
    history: string[]
}

impl Calculator {
    fn new(): Calculator
    fn add(value: float): Calculator
    fn subtract(value: float): Calculator
    fn multiply(value: float): Calculator
    fn divide(value: float): Result<Calculator, string>
    fn clear(): Calculator
    fn get_value(): float
    fn get_history(): string[]
}

test.spec("Stateful Calculator", || {
    test.it("should start with zero", || {
        let calc = Calculator.new()
        test.assert_eq(calc.get_value(), 0.0)
    })
    
    test.it("should chain operations", || {
        let calc = Calculator.new()
            .add(10.0)
            .subtract(3.0)
            .multiply(2.0)
        
        test.assert_eq(calc.get_value(), 14.0)
    })
    
    test.it("should track operation history", || {
        let calc = Calculator.new()
            .add(5.0)
            .multiply(2.0)
        
        let history = calc.get_history()
        test.assert_eq(history.len(), 2)
        test.assert(history[0].contains("+ 5.0"))
        test.assert(history[1].contains("* 2.0"))
    })
    
    test.it("should handle division by zero gracefully", || {
        let calc = Calculator.new().add(10.0)
        let result = calc.divide(0.0)
        
        match result {
            Ok(_) => test.fail("Should have returned an error"),
            Err(error) => {
                test.assert(error.contains("zero"))
                // Original calculator should be unchanged
                test.assert_eq(calc.get_value(), 10.0)
            }
        }
    })
})
```

## Step 7: Implement the Stateful Calculator

```boba
// Start with stub implementations (Red phase)
impl Calculator {
    fn new(): Calculator {
        return Calculator {
            current_value: 0.0,
            history: []
        }
    }
    
    fn add(value: float): Calculator {
        let new_value = self.current_value + value
        let operation = f"+ {value} = {new_value}"
        
        return Calculator {
            current_value: new_value,
            history: [...self.history, operation]
        }
    }
    
    fn subtract(value: float): Calculator {
        let new_value = self.current_value - value
        let operation = f"- {value} = {new_value}"
        
        return Calculator {
            current_value: new_value,
            history: [...self.history, operation]
        }
    }
    
    fn multiply(value: float): Calculator {
        let new_value = self.current_value * value
        let operation = f"* {value} = {new_value}"
        
        return Calculator {
            current_value: new_value,
            history: [...self.history, operation]
        }
    }
    
    fn divide(value: float): Result<Calculator, string> {
        if value == 0.0 {
            return Err("Cannot divide by zero")
        }
        
        let new_value = self.current_value / value
        let operation = f"/ {value} = {new_value}"
        
        let new_calc = Calculator {
            current_value: new_value,
            history: [...self.history, operation]
        }
        
        return Ok(new_calc)
    }
    
    fn clear(): Calculator {
        return Calculator {
            current_value: 0.0,
            history: ["Cleared"]
        }
    }
    
    fn get_value(): float {
        return self.current_value
    }
    
    fn get_history(): string[] {
        return self.history
    }
}
```

## Step 9: Testing Patterns and Best Practices

### Setup and Teardown

```boba
test.spec("Calculator with Common Setup", || {
    fn setup_calculator(): Calculator {
        return Calculator.new()
            .add(10.0)
            .multiply(2.0)  // starts with 20.0
    }
    
    test.it("should start from setup state", || {
        let calc = setup_calculator()
        test.assert_eq(calc.get_value(), 20.0)
    })
    
    test.it("should add to setup state", || {
        let calc = setup_calculator().add(5.0)
        test.assert_eq(calc.get_value(), 25.0)
    })
})
```

### Parameterized Tests

```boba
test.spec("Calculator Addition - Multiple Cases", || {
    let test_cases = [
        (0.0, 0.0, 0.0),
        (1.0, 1.0, 2.0),
        (-1.0, 1.0, 0.0),
        (5.5, 2.3, 7.8)
    ]
    
    for (a, b, expected) in test_cases {
        test.it(f"should add {a} + {b} = {expected}", || {
            let calc = Calculator.new().add(a).add(b)
            let result = calc.get_value()
            
            // Handle floating point comparison
            test.assert(abs(result - expected) < 0.0001)
        })
    }
})

fn abs(x: float): float {
    return if x < 0.0 { -x } else { x }
}
```

### Mock Objects and Dependency Injection

```boba
// Interface for external dependencies
trait Logger {
    fn log(message: string)
}

struct Calculator {
    current_value: float,
    history: string[],
    logger: Logger
}

// Mock logger for testing
struct MockLogger {
    logged_messages: string[]
}

impl Logger for MockLogger {
    fn log(message: string) {
        self.logged_messages.push(message)
    }
}

test.spec("Calculator with Logging", || {
    test.it("should log operations", || {
        let mock_logger = MockLogger { logged_messages: [] }
        let calc = Calculator.new_with_logger(mock_logger)
            .add(5.0)
        
        test.assert_eq(mock_logger.logged_messages.len(), 1)
        test.assert(mock_logger.logged_messages[0].contains("Added 5.0"))
    })
})
```

## Benefits of TDD in Boba

1. **Better Design**: Tests force you to think about interfaces first
2. **Error Handling**: Result types make error scenarios explicit
3. **Type Safety**: Compiler catches type mismatches in tests
4. **Documentation**: Tests serve as executable specifications
5. **Refactoring Confidence**: Safe to change implementation

## Common TDD Patterns

- **AAA Pattern**: Arrange, Act, Assert
- **Given-When-Then**: Test scenario structure
- **Test Doubles**: Mocks, stubs, and fakes for dependencies
- **Red-Green-Refactor**: Core TDD cycle

Following TDD in Boba leads to more reliable, well-designed code with comprehensive test coverage!
